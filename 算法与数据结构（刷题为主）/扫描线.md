````
# 扫描线核心思路

1.  确定扫描方向 - 通常是 X 轴从左到右，或按时间线推进
2.  定义事件点 - 什么位置/时刻需要处理（线段起点、终点、交点等）
3.  维护活跃状态 - 扫描线当前"穿过"了哪些东西
4.  在事件点更新状态并产生结果

---

## 常见应用场景

| 场景 | 扫什么 | 为什么好用 |
| :--- | :--- | :--- |
| 矩形面积并（Union Area） | 沿 X 轴扫，每个矩形的左右边是事件 | 把 2D 问题降成 1D：每次只算当前被覆盖的 Y 区间长度 × X 增量 |
| 线段交点检测 | 沿 X 轴扫，线段端点是事件 | 只需检查 Y 坐标相邻的线段是否相交，而不是 O(n²) 暴力两两比对 |
| 区间最大重叠 | 沿时间线扫，每个区间的开始/结束是事件 | 会议室预定、最大并发连接数等问题，O(n log n) 解决 |
| 多边形填充（图形学） | 逐行扫描，边与扫描线的交点是事件 | 判断像素在多边形内部还是外部 |
| 天际线问题 | 沿 X 轴扫，建筑物左右边是事件 | 维护当前最高高度，高度变化时输出关键点 |

---

## 一个简单例子
计算多个区间的最大重叠数：
- 把每个区间拆成 `(起点, +1)` 和 `(终点, -1)` 两个事件
- 按坐标排序，扫描过去，维护当前重叠计数，取最大值

```cpp
// 区间最大重叠 - 扫描线思路
// events: [(x, +1), (x, -1), ...] 排序后扫描
for (event in sorted_events) {
    count += event.delta;
    max_overlap = max(max_overlap, count);
}
````

## 一句话总结

扫描线本质是 "只关心变化点的遍历" - 把连续问题离散化为事件序列处理，通常能将 O (n²) 暴力降为 O (n log n)。当你遇到 "在某个维度上有起点和终点" 的问题时，就值得考虑扫描线。

观察图发现有几条线段 我们线段树维护的是这个线段

第三个是线段的真实覆盖长度

# 2求矩形的并的长度
![[Pasted image 20260515125424.png]]
思路与矩形的并其实是差不多的 先求y方向上的周长再去求x方向上的周长
用一个y来记录来到某根线的时候从最顶部到最底部值的变化 这个变化量（绝对值） 的和就是周长的和
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
typedef long long ll;
typedef struct segment{
    int l,r;//代表线段的下端点和上端点
    int x;
    int k;//这个反映的是出边还是入边
}seg;
seg segs[N<<1];
int n;
int yy[N<<1];
bool cmp(seg a,seg b)
{
    return a.x<b.x;
}
ll sum2[N<<4];//这个是某一范围的总长度 这个长度是确定的
ll sum1[N<<4];//这个是覆盖的长度 
int tag[N<<4];//这个是覆盖的次数
void up(int i)
{
    if(!tag[i])
    {
        sum1[i]=sum1[i<<1]+sum1[i<<1|1];
    }
}
void build(int i,int l,int r)
{
    if(l==r)
    {
        // cout<<"i"<<" "<<l<<" "<<r<<" "<<endl;; 
        sum2[i]=yy[r+1]-yy[l];
        return ;
    }
    int midd=(l+r)/2;
    build(i<<1,l,midd);
    build(i<<1|1,midd+1,r);
    sum2[i]=sum2[i<<1]+sum2[i<<1|1];
}
void update(int jobl,int jobr,int l,int r,int i,int k)
{
    //节点的[l,r]范围其实就是维护的线段的范围
    if(jobl<=l&&r<=jobr)
    {
        tag[i]+=k;
        if(!tag[i])
        {
            sum1[i]=sum1[i<<1]+sum1[i<<1|1];
        }
        else
        {
            sum1[i]=sum2[i];
        }
        return ;
    }
    int midd=(l+r)/2;
    if(jobl<=midd)
    update(jobl,jobr,l,midd,i<<1,k);
    if(jobr>midd)
    update(jobl,jobr,midd+1,r,i<<1|1,k);
    up(i);
}
int main()
{
    cin>>n;
    int x1,y1,x2,y2;
    for(int i=1;i<=n;i++)
    {
        cin>>x1>>y1>>x2>>y2;
        segs[i]={y1,y2,x1,1};
        segs[i+n]={y1,y2,x2,-1};
        yy[i]=y1;
        yy[i+n]=y2;

    }
    n=n*2;
    sort(segs+1,segs+1+n,cmp);
    sort(yy+1,yy+1+n);
    int tot=unique(yy+1,yy+1+n)-yy-2;
    build(1,1,tot);
    ll ans=0;
    // cout<<tot<<endl;
    // for(int i=1;i<=n;i++)
    // {
    //     cout<<yy[i]<<" ";
    // }cout<<endl;
    for(int i=1;i<n;i++)
    {
        y1=lower_bound(yy+1,yy+tot+2,segs[i].l)-yy;
        y2=lower_bound(yy+1,yy+tot+2,segs[i].r)-yy;
        update(y1,y2-1,1,tot,1,segs[i].k);
        ans=ans+sum1[1]*(segs[i+1].x-segs[i].x);
    }
    cout<<ans<<endl;
}
```
