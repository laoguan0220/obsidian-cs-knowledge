分析:
1.进行单点修改的时候 不需要懒标记 这次每次更新都是到叶子结点
2.每次更新的时候 都会有logn级别的空间被创建 其它的老节点被复用 逻辑上相当于n棵线段树 
3.cnt来记录下来所有这些线段树的节点数量信息
root数组记录下来第i棵线段树的根节点是哪个节点
arr初始数组
value数组记录下来节点i的值 这道题只有叶子节点是有这个信息的 所以甚至都不需要进行up操作向上进行更新
节点的数量我们可以认为最多是 4*{n}+nlogn

4.注意  所有的线段树  长得其实是一样的 也就是说每个节点的范围l _r 其实都是一样的

5.build方法 返回的是头节点的编号 
6.左孩子右孩子我们尽量给写成lch 和rch 
7.update方法也是要返回建立好的新版本线段树的根节点的(int)类型 递归地去调用

#include<bits/stdc++.h>

using namespace std;

int n,m;

const int N=1e6+5;

const int MAXT=(N<<2)+N*22;

int root[N<<2];//第i棵线段树 根节点的编号

int arr[N];//这个是初始的arr数组

int value[MAXT];//第i个节点的值

int lch[MAXT];//节点的左孩子右孩子

int rch[MAXT];

int cnt=0;

int build(int l,int r)

{

    int rf=++cnt;

    if(l==r)

    {

        value[rf]=arr[l];

        return rf;

    }

    int midd=(l+r)/2;

    lch[rf]=build(l,midd);

    rch[rf]=build(midd+1,r);

    return rf;

}

int update(int rt,int p,int c,int l,int r)

{

    //rt是基于v版本的 我们要构建出来一个新的版本 返回新版本的根节点

    int rf=++cnt;

    if(l==r)

    {

        value[rf]=c;

    }

    else

    {

        int midd=(l+r)/2;

        lch[rf]=lch[rt];

        rch[rf]=rch[rt];

        if(p<=midd)

        {

            lch[rf]=update(lch[rt],p,c,l,midd);

        }

        else

        {

            rch[rf]=update(rch[rt],p,c,midd+1,r);

        }

    }

    return rf;

}

int query(int rt,int p,int l,int r)

{

    if(l==r)

    {

        return value[rt];

    }

    int midd=(l+r)/2;

    if(p<=midd)

    {

        return query(lch[rt],p,l,midd);

    }

    else

    {

        return query(rch[rt],p,midd+1,r);

    }

}

int main()

{

    ios::sync_with_stdio(false);

cin.tie(0);

    cin>>n>>m;

    for(int i=1;i<=n;i++)

    {

        cin>>arr[i];

    }

    root[0]=build(1,n);

    int v,op,p,c;

    for(int i=1;i<=m;i++)

    {

        cin>>v>>op>>p;

        if(op==1)

        {

            cin>>c;

            root[i]=update(root[v],p,c,1,n);

        }

        else

        {

            cout<<query(root[v],p,1,n)<<endl;

            root[i]=root[v];

        }

    }

}
