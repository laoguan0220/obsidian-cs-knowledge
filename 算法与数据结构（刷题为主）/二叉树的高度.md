## 二叉树的高度

### 什么是二叉树的高度？

**二叉树的高度**（Height）是指从根节点到最远叶子节点的最长路径上的节点数（或边数）。 一般来讲高度指的是节点的数量 

- **节点数定义**：根节点高度为1，空树高度为0
- **边数定义**：根节点高度为0，空树高度为-1

通常我们使用**节点数定义**。

### 二叉树高度的应用

1. **平衡二叉树判断**：AVL树要求左右子树高度差不超过1
2. **二叉搜索树性能分析**：高度影响搜索、插入、删除的时间复杂度
3. **堆排序**：堆的高度影响建堆和调整的效率
4. **树的遍历**：递归深度受高度限制
5. **空间复杂度分析**：递归调用栈的深度等于树的高度
6. **决策树剪枝**：限制树的最大深度防止过拟合

## C++ 演示代码

```cpp
#include <iostream>
#include <algorithm>
#include <queue>
#include <cmath>
using namespace std;

// 二叉树节点定义
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

class BinaryTree {
private:
    TreeNode* root;

    // 递归计算高度
    int getHeightRecursive(TreeNode* node) {
        if (node == nullptr) return 0;
        int leftHeight = getHeightRecursive(node->left);
        int rightHeight = getHeightRecursive(node->right);
        return max(leftHeight, rightHeight) + 1;
    }

    // 层序遍历计算高度（非递归）
    int getHeightIterative(TreeNode* root) {
        if (root == nullptr) return 0;
        
        queue<TreeNode*> q;
        q.push(root);
        int height = 0;
        
        while (!q.empty()) {
            int levelSize = q.size();
            height++;
            
            for (int i = 0; i < levelSize; i++) {
                TreeNode* current = q.front();
                q.pop();
                
                if (current->left) q.push(current->left);
                if (current->right) q.push(current->right);
            }
        }
        return height;
    }

    // 判断是否为平衡二叉树
    bool isBalancedHelper(TreeNode* node, int& height) {
        if (node == nullptr) {
            height = 0;
            return true;
        }
        
        int leftHeight = 0, rightHeight = 0;
        bool leftBalanced = isBalancedHelper(node->left, leftHeight);
        bool rightBalanced = isBalancedHelper(node->right, rightHeight);
        
        height = max(leftHeight, rightHeight) + 1;
        
        return leftBalanced && rightBalanced && 
               abs(leftHeight - rightHeight) <= 1;
    }

    // 释放内存
    void deleteTree(TreeNode* node) {
        if (node == nullptr) return;
        deleteTree(node->left);
        deleteTree(node->right);
        delete node;
    }

public:
    BinaryTree() : root(nullptr) {}
    
    ~BinaryTree() {
        deleteTree(root);
    }

    // 插入节点（构建二叉搜索树）
    void insert(int val) {
        if (root == nullptr) {
            root = new TreeNode(val);
            return;
        }
        
        TreeNode* current = root;
        while (true) {
            if (val < current->val) {
                if (current->left == nullptr) {
                    current->left = new TreeNode(val);
                    break;
                }
                current = current->left;
            } else {
                if (current->right == nullptr) {
                    current->right = new TreeNode(val);
                    break;
                }
                current = current->right;
            }
        }
    }

    // 获取树的高度（递归）
    int getHeight() {
        return getHeightRecursive(root);
    }

    // 获取树的高度（非递归）
    int getHeightNonRecursive() {
        return getHeightIterative(root);
    }

    // 判断是否为平衡二叉树
    bool isBalanced() {
        int height = 0;
        return isBalancedHelper(root, height);
    }

    // 打印树的结构（简化版）
    void printTree() {
        if (root == nullptr) {
            cout << "空树" << endl;
            return;
        }
        
        queue<TreeNode*> q;
        q.push(root);
        
        while (!q.empty()) {
            int levelSize = q.size();
            
            for (int i = 0; i < levelSize; i++) {
                TreeNode* current = q.front();
                q.pop();
                
                if (current) {
                    cout << current->val << " ";
                    q.push(current->left);
                    q.push(current->right);
                } else {
                    cout << "N ";
                }
            }
            cout << endl;
        }
    }
};

// 演示函数
void demonstrateHeight() {
    cout << "=== 二叉树高度演示 ===" << endl << endl;
    
    // 创建示例树
    BinaryTree tree;
    
    // 插入节点构建二叉搜索树
    int values[] = {5, 3, 7, 2, 4, 6, 8, 1, 9};
    for (int val : values) {
        tree.insert(val);
    }
    
    cout << "树的结构（层序遍历）：" << endl;
    tree.printTree();
    
    cout << "\n树的高度（递归计算）：" << tree.getHeight() << endl;
    cout << "树的高度（非递归计算）：" << tree.getHeightNonRecursive() << endl;
    cout << "是否为平衡二叉树：" << (tree.isBalanced() ? "是" : "否") << endl;
    
    // 创建一个不平衡的树
    BinaryTree unbalancedTree;
    for (int i = 1; i <= 5; i++) {
        unbalancedTree.insert(i);
    }
    
    cout << "\n\n不平衡树的结构：" << endl;
    unbalancedTree.printTree();
    cout << "高度：" << unbalancedTree.getHeight() << endl;
    cout << "是否为平衡二叉树：" << (unbalancedTree.isBalanced() ? "是" : "否") << endl;
}

// 高度在算法中的应用演示
void demonstrateApplications() {
    cout << "\n\n=== 高度在算法中的应用 ===" << endl;
    
    // 1. 时间复杂度分析
    cout << "\n1. 时间复杂度分析：" << endl;
    cout << "   二叉搜索树搜索时间复杂度：O(h)，其中h为树的高度" << endl;
    cout << "   平衡二叉树：O(log n)" << endl;
    cout << "   退化为链表：O(n)" << endl;
    
    // 2. 空间复杂度分析
    cout << "\n2. 空间复杂度分析：" << endl;
    cout << "   递归遍历的空间复杂度：O(h)" << endl;
    cout << "   平衡二叉树递归深度：O(log n)" << endl;
    cout << "   退化为链表递归深度：O(n)" << endl;
    
    // 3. 平衡二叉树判断
    cout << "\n3. 平衡二叉树判断：" << endl;
    cout << "   AVL树要求：|height(left) - height(right)| <= 1" << endl;
    cout << "   用于保证树的平衡性，确保操作效率" << endl;
}

int main() {
    demonstrateHeight();
    demonstrateApplications();
    
    return 0;
}
```

## 关键点总结

1. **高度定义**：根节点到最远叶子节点的路径长度
2. **计算方法**：递归（后序遍历）或非递归（层序遍历）
3. **时间复杂度**：O(n)，n为节点数
4. **空间复杂度**：递归O(h)，非递归O(w)，w为最大宽度
5. **主要应用**：平衡判断、性能分析、树的操作优化
