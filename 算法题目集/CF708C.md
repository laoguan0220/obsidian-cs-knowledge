题目6
哪些点可以改造成重心
给定一棵n个点的树，你可以随便删掉一条边，然后随便加上一条边
通过这种方式可以让很多点变成重心
打印所有能变成重心的点
测试链接 : https://www.luogu.com.cn/problem/CF708C
测试链接 : https://codeforces.com/problemset/problem/708/C
分析：[[换根dp]]
1.由瓶颈分析法可以知道 问题是如何将较大的联通块变成较小的 down up 分别表示两大联通块里的 能删除子树的最大大小 但是 得<=n/2  
2.为了快速得到一个节点的兄弟节点的down数组的最大值 我们需要记录下来最大值 和次大值 
同时记录下来 最大值是来自于哪个儿子 ，当处理最大值儿子的时候 为次大值 当处理非最大值儿子的时候 结果肯定是最大值 ，同时这样写可以处理有多个最大值的情况 
当然也可以维护最大值的次数去写
`down[u]` 里面说“可以切走 `u` 自己这棵子树”，不是说把节点 `u` 删除掉，而是说**删除 `u` 和它父亲之间的那条边**，从而把以 `u` 为根的整棵子树切下来
```cpp
#include <bits/stdc++.h>

using namespace std;

  

#define endl "\n"

  

const int N = 4e5 + 5;

  

int n;

vector<int> adj[N];

  

int down[N], up[N], sz[N], fa[N];

int fmax1[N], fmax2[N], fson[N];

  

void dfs(int rt, int rf)

{

    fa[rt] = rf;

    sz[rt] = 1;

  

    for(int to : adj[rt])

    {

        if(to == rf) continue;

  

        dfs(to, rt);

        sz[rt] += sz[to];

    }

  

    down[rt] = (sz[rt] <= n / 2 ? sz[rt] : 0);

  

    for(int to : adj[rt])

    {

        if(to == rf) continue;

  

        down[rt] = max(down[rt], down[to]);

  

        int mt = down[to];

  

        if(mt > fmax1[rt])

        {

            fmax2[rt] = fmax1[rt];

            fmax1[rt] = mt;

            fson[rt] = to;

        }

        else if(mt > fmax2[rt])

        {

            fmax2[rt] = mt;

        }

    }

}

  

void dfs1(int rt, int rf)

{

    for(int to : adj[rt])

    {

        if(to == rf) continue;

  

        if(n - sz[to] <= n / 2)

        {

            up[to] = n - sz[to];

        }

        else

        {

            up[to] = up[rt];

  

            if(fson[rt] == to)

            {

                up[to] = max(up[to], fmax2[rt]);

            }

            else

            {

                up[to] = max(up[to], fmax1[rt]);

            }

        }

  

        dfs1(to, rt);

    }

}

  

int main()

{

    ios::sync_with_stdio(false);

    cin.tie(nullptr);

  

    cin >> n;

  

    for(int i = 1; i < n; i++)

    {

        int u, v;

        cin >> u >> v;

  

        adj[u].push_back(v);

        adj[v].push_back(u);

    }

  

    dfs(1, 0);

  

    up[1] = 0;

    dfs1(1, 0);

  

    for(int u = 1; u <= n; u++)

    {

        bool ok = true;

  

        int parentPart = n - sz[u];

  

        if(parentPart > n / 2)

        {

            if(parentPart - up[u] > n / 2)

            {

                ok = false;

            }

        }

  

        for(int v : adj[u])

        {

            if(fa[v] == u)

            {

                if(sz[v] > n / 2)

                {

                    if(sz[v] - down[v] > n / 2)

                    {

                        ok = false;

                    }

                }

            }

        }

  

        cout << (ok ? 1 : 0) << " ";

    }

  

    cout << endl;

  

    return 0;

}
```
