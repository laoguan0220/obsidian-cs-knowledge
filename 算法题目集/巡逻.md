[[树的直径]]
1.先两遍dfs求出来直径 建立好last信息和dist信息 然后将直径上的边权设置为-1
然后再跑一遍树形dp 得到第二条直径 利用上面那个公式得到结果
```cpp
#include<bits/stdc++.h>

using namespace std;

const int N=1e5+10;

typedef struct node{

    int v;

    int w;

}node;

vector<node>adj[N];

int dist[N];

int dist1[N];

int ans[N];

int last[N];

// int cnt[N][2];//cnt[i][0]表示父亲节点到它是第几条边 、

int farnode;

int maxlen=-1;

int n,k;

void dfs(int rt,int rf,int ww)

{

    dist[rt]=ww;

    last[rt]=rf;

    if(ww>maxlen)

    {

        maxlen=ww;

        farnode=rt;

    }

    for(int i=0;i<adj[rt].size();i++)

    {

        if(adj[rt][i].v!=rf)

        {

            dfs(adj[rt][i].v,rt,ww+1);

        }

    }

}

void dp(int rt,int rf)

{

    for(int i=0;i<adj[rt].size();i++)

    {

        if(adj[rt][i].v!=rf)

        {

            dp(adj[rt][i].v,rt);

            ans[rt]=max({ans[rt],ans[adj[rt][i].v],dist1[rt]+dist1[adj[rt][i].v]+adj[rt][i].w});

            dist1[rt]=max(dist1[rt],dist1[adj[rt][i].v]+adj[rt][i].w);

        }

    }

}

int main()

{

    cin>>n>>k;

    for(int i=1;i<n;i++)

    {

        int a,b;

        cin>>a>>b;

        adj[a].push_back({b,1});

        adj[b].push_back({a,1});

    }

    dfs(1,0,0);

    maxlen=-1;

    int start=farnode;

    int res=0;

    dfs(start,0,0);

    // cout<<maxlen<<endl;

    if(k==1)

    {

        res=2*(n-1)+1-maxlen;

        cout<<res<<endl;

        return 0;

    }

    for(int i=farnode;i!=start;i=last[i])

    {

        //adj[i][j]

        int j=last[i];

        for(int k=0;k<adj[i].size();k++)

        {

            if(adj[i][k].v==j)

            {

                adj[i][k].w=-1;

                break;

            }

        }

        for(int k=0;k<adj[j].size();k++)

        {

            if(adj[j][k].v==i)

            {

                adj[j][k].w=-1;

                break;

            }

        }

    }

    dp(1,0);

    // cout<<ans[1]<<endl;

    res=max(res,2*(n-1)-maxlen-ans[1]+2);

    cout<<res<<endl;

}
```
