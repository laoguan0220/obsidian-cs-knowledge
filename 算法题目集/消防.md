[[树的直径]]
对于一个节点 如果是非负权树 距离它的最远的点 一定是直径的某端 
找到树的重心那根链 遍历所有的符合条件的线段吗？再用单调队列优化将O$(n^2)$ 变成$O(n)$
但是这里一定要注意 对于消防站单个点来讲 它的最大值确实只需要考虑两端 但是你对消防站连续的几个点来讲 到它最远的点可不一定在直径的两端 这个要注意 所以我们需要从头开始遍历 dist数组 表示从这个节点开始 不经过链上的节点可以到达的最大距离 
这里我们需要考虑 遍历所有的单调队列的开头 然后答案 由在单调队列里面的那几个数据的最大子树 以及两端到两边的距离所决定的
```cpp
#include<bits/stdc++.h>

using namespace std;

int n,s;

typedef struct node{

    int v;

    int w;

}node;

const int N=3e5+10;

vector<node>adj[N];

int farnode,start,maxlen=-1;

int dist[N];

int maxdist[N];

int last[N];

// int pred[N];//这个数组记录的是 直径的某个节点到上一个节点的路径长度是多少

void dfs(int rt,int rf,int ww)

{

    dist[rt]=ww;

    last[rt]=rf;

    if(ww>maxlen)

    {

        maxlen=ww;

        farnode=rt;

    }

    for(int i=0;i<adj[rt].size();i++)

    {

        if(adj[rt][i].v!=rf)

        {

            dfs(adj[rt][i].v,rt,ww+adj[rt][i].w);

            // pred[adj[rt][i].v]=adj[rt][i].w;          

        }

    }

}

bool visited[N];

void dfs1(int rt,int rf)

{

    for(int i=0;i<adj[rt].size();i++)

    {

        if(adj[rt][i].v!=rf)

        {

            dfs1(adj[rt][i].v,rt);

            if(!visited[adj[rt][i].v])

            maxdist[rt]=max(maxdist[rt],maxdist[adj[rt][i].v]+adj[rt][i].w);

        }

    }

}

int queue1[N];

int main()

{

    cin>>n>>s;

    for(int i=1;i<n;i++)

    {

        int u,v,w;

        cin>>u>>v>>w;

        adj[u].push_back({v,w});

        adj[v].push_back({u,w});

    }

    dfs(1,0,0);

    start=farnode;

    maxlen=-1;

    dfs(start,0,0);

    //这个时候farnode就是直径 最远点

    for(int i=farnode;i!=start;i=last[i])

    {

        visited[i]=true;

    }

    visited[start]=true;

    dfs1(start,0);

//我们其实可以从farnode其实走 也是可以的

    //我们的范围认为 [i,r]

    //int queue1[N];

    queue1[1]=farnode;

    int h=1,t=1;

    int mincnt=INT_MAX;

    for(int l=farnode,r=farnode;l!=0;l=last[l])

    {

        while(last[r]!=0&&dist[l]-dist[last[r]]<=s)

        {

            while(h<=t&&maxdist[last[r]]>=maxdist[queue1[t]])

            {

                t--;

            }

            queue1[++t]=last[r];

            r=last[r];

        }

        mincnt=min(mincnt,max({dist[r],dist[farnode]-dist[l],maxdist[queue1[h]]}));

        if(queue1[h]==l)

        {

            h++;

        }

    }

    cout<<mincnt<<endl;

}
//单调队列 单调递减 维护最有可能的答案 某一范围内的
```
